# Лабораторная работа №3: Реализация шаблонного класса Vector

## 🎯 Цель работы
1. Реализовать шаблонный класс `Vector<T>` — аналог `std::vector` — с нуля.
2. Освоить управление динамической памятью: выделение, расширение, освобождение.
3. Закрепить «правило трёх» (деструктор, конструктор копирования, оператор присваивания).
4. Провести амортизированный анализ операции `push_back`.

## 📋 Вариант задания
**Базовый уровень, Вариант №4**
*   **Конструкторы:** по умолчанию, копирования.
*   **Обязательные методы:** `push_back`, `pop_back`, `size`, `reserve`.
*   **Дополнительно:** реализован деструктор и оператор присваивания копированием (для соблюдения «правила трёх» и предотвращения утечек памяти).

## 📚 Теоретическая часть
*   **Динамический массив** — структура данных в куче (heap), которая может изменять свой размер во время выполнения.
*   **Ёмкость (capacity)** — сколько памяти реально выделено. **Размер (size)** — сколько элементов реально хранится.
*   **Правило трёх:** если класс управляет ресурсами (динамической памятью), ему необходимы: деструктор, конструктор копирования и оператор присваивания копированием.
*   **Амортизированная сложность `push_back`:** Благодаря стратегии удвоения емкости (`cap * 2`), редкие операции копирования `O(n)` окупаются частыми дешевыми операциями, что дает амортизированное время `O(1)`.

## 💻 Листинг программы (main.cpp)

```cpp
#include <iostream>
#include <stdexcept>
#include <utility> // для std::swap

using namespace std;

template <typename T>
class Vector {
private:
    T* data;    // Указатель на динамический массив
    int sz;     // Текущий размер (количество элементов)
    int cap;    // Ёмкость (выделенная память)

public:
    // 1. Конструктор по умолчанию
    Vector() : data(nullptr), sz(0), cap(0) {
        cout << "Вызван конструктор по умолчанию." << endl;
    }

    // 2. Конструктор копирования (глубокое копирование)
    Vector(const Vector& other) : sz(other.sz), cap(other.cap) {
        cout << "Вызван конструктор копирования (глубокое копирование)." << endl;
        data = new T[cap];
        for (int i = 0; i < sz; i++) {
            data[i] = other.data[i];
        }
    }

    // 3. Оператор присваивания копированием (защита от самоприсваивания)
    Vector& operator=(const Vector& other) {
        if (this != &other) { // Проверка на самоприсваивание
            // Временный объект для безопасного копирования (copy-and-swap idiom)
            Vector temp(other);
            
            // Обмен содержимым
            swap(data, temp.data);
            swap(sz, temp.sz);
            swap(cap, temp.cap);
        }
        return *this;
    }

    // 4. Деструктор
    ~Vector() {
        delete[] data;
        cout << "Вызван деструктор (память освобождена)." << endl;
    }

    // Метод reserve (выделение памяти)
    void reserve(int newCap) {
        if (newCap <= cap) return; // Если запрошено меньше, ничего не делаем

        T* newData = new T[newCap];
        for (int i = 0; i < sz; i++) {
            newData[i] = data[i]; // Копируем старые элементы
        }
        delete[] data; // Освобождаем старую память
        data = newData;
        cap = newCap;
    }

    // Метод push_back (амортизированно O(1))
    void push_back(const T& value) {
        if (sz >= cap) {
            // Если места нет, расширяем ёмкость (в 2 раза или до 1)
            int newCap = (cap == 0) ? 1 : cap * 2;
            reserve(newCap);
        }
        data[sz++] = value;
    }

    // Метод pop_back
    void pop_back() {
        if (sz > 0) {
            sz--;
        } else {
            cout << "Ошибка: попытка удалить элемент из пустого вектора!" << endl;
        }
    }

    // Метод size
    int size() const { return sz; }
    int capacity() const { return cap; }

    // Дополнительный метод для демонстрации содержимого
    void print() const {
        cout << "[";
        for (int i = 0; i < sz; i++) {
            cout << data[i];
            if (i < sz - 1) cout << ", ";
        }
        cout << "]";
    }
};

// ==========================================
// Главная функция (Демонстрация)
// ==========================================
int main() {
    setlocale(LC_ALL, "ru");

    cout << "--- 1. Создание вектора и использование push_back ---" << endl;
    Vector<int> v1;
    for (int i = 1; i <= 5; i++) {
        v1.push_back(i * 10);
    }
    cout << "Вектор v1: ";
    v1.print();
    cout << " (Размер: " << v1.size() << ", Ёмкость: " << v1.capacity() << ")\n\n";

    cout << "--- 2. Использование метода reserve ---" << endl;
    Vector<int> v2;
    v2.reserve(10); // Сразу выделяем память под 10 элементов
    cout << "После reserve(10): Размер: " << v2.size() << ", Ёмкость: " << v2.capacity() << endl;
    v2.push_back(99);
    cout << "Вектор v2: ";
    v2.print();
    cout << "\n\n";

    cout << "--- 3. Демонстрация pop_back ---" << endl;
    v1.pop_back(); // Удаляем последний элемент (50)
    cout << "Вектор v1 после pop_back: ";
    v1.print();
    cout << " (Размер: " << v1.size() << ")\n\n";

    cout << "--- 4. Демонстрация конструктора копирования ---" << endl;
    Vector<int> v3 = v1; // Вызов конструктора копирования
    cout << "Вектор v3 (копия v1): ";
    v3.print();
    cout << "\n\n";

    cout << "--- 5. Демонстрация оператора присваивания ---" << endl;
    Vector<int> v4;
    v4 = v2; // Присваивание
    cout << "Вектор v4 (присвоен из v2): ";
    v4.print();
    cout << "\n\n";

    cout << "--- Конец программы (вызов деструкторов) ---" << endl;
    return 0;
}
